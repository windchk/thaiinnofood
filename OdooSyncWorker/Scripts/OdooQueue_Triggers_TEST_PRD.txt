Odoo queue triggers for Production Order and A/R Reserve Invoice
================================================================

Required queue migration
------------------------
Run Scripts/INT_OdooQueue_Setup.txt first so TEMP_API.dbo.INT_OdooQueue
contains SiteId, CompanyName, and SapDatabaseName.

Company mapping
---------------
TEST-TIF = TEST_INTERFACE
TEST-STL = TEST_STL_ODOO
PRD-TIF  = TIF_GOLIVE
PRD-STL  = SBO_STL_GOLIVE

Queue CompanyName stores TIF/STL. The outgoing Odoo field companyName uses
the resolved SapDatabaseName shown on the right side of this mapping.

Events
------
SalesOrder and ItemMaster:
- Temporarily disabled.
- Legacy TRG_ORDR_ToOdooQueue and TRG_OITM_ToOdooQueue are disabled below
  when this script is run in the TIF TEST/PRD databases.

ProductionOrder:
- RELEASED when OWOR.Status enters R
- CANCEL when OWOR.Status enters C
- Worker sends RELEASED as docState = release and CANCEL as docState = cancel.
- Production Order trigger is installed only in TIF databases.

ARReserveInvoice:
- ADD when an OINV row with isIns = Y is inserted
- CANCEL when the original reserve invoice changes CANCELED to Y
- Worker sends ADD as docState = release and CANCEL as docState = cancel.
- Updates unrelated to Add/Cancel are not queued.
- Cancellation documents where CANCELED = C are not queued as ADD.


TEST / TIF
----------

USE [TEST_INTERFACE];
GO

IF OBJECT_ID(N'dbo.TRG_ORDR_ToOdooQueue', N'TR') IS NOT NULL
    DISABLE TRIGGER dbo.TRG_ORDR_ToOdooQueue ON dbo.ORDR;
GO

IF OBJECT_ID(N'dbo.TRG_OITM_ToOdooQueue', N'TR') IS NOT NULL
    DISABLE TRIGGER dbo.TRG_OITM_ToOdooQueue ON dbo.OITM;
GO

CREATE OR ALTER TRIGGER dbo.TRG_OWOR_ToOdooQueue
ON dbo.OWOR
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO TEMP_API.dbo.INT_OdooQueue
    (
        ObjectType,
        ObjectKey,
        ActionType,
        SiteId,
        CompanyName,
        Status,
        RetryCount,
        CreateDate
    )
    SELECT
        'ProductionOrder',
        CAST(I.DocEntry AS NVARCHAR(50)),
        CASE I.Status WHEN 'R' THEN 'RELEASED' ELSE 'CANCEL' END,
        'TEST-TIF',
        'TIF',
        'N',
        0,
        GETDATE()
    FROM inserted I
    LEFT JOIN deleted D
        ON I.DocEntry = D.DocEntry
    WHERE
        (
            (
                I.Status = 'R'
                AND ISNULL(D.Status, '') <> 'R'
            )
            OR
            (
                I.Status = 'C'
                AND ISNULL(D.Status, '') <> 'C'
            )
        )
        AND NOT EXISTS
        (
            SELECT 1
            FROM TEMP_API.dbo.INT_OdooQueue Q
            WHERE Q.ObjectType COLLATE DATABASE_DEFAULT = 'ProductionOrder'
              AND Q.ObjectKey COLLATE DATABASE_DEFAULT = CAST(I.DocEntry AS NVARCHAR(50))
              AND Q.ActionType COLLATE DATABASE_DEFAULT = CASE I.Status WHEN 'R' THEN 'RELEASED' ELSE 'CANCEL' END
              AND Q.SiteId COLLATE DATABASE_DEFAULT IN ('TEST-TIF', 'TEST')
              AND Q.CompanyName COLLATE DATABASE_DEFAULT = 'TIF'
              AND Q.Status COLLATE DATABASE_DEFAULT IN ('N', 'P')
        );
END;
GO

CREATE OR ALTER TRIGGER dbo.TRG_OINV_Reserve_ToOdooQueue
ON dbo.OINV
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO TEMP_API.dbo.INT_OdooQueue
    (
        ObjectType,
        ObjectKey,
        ActionType,
        SiteId,
        CompanyName,
        Status,
        RetryCount,
        CreateDate
    )
    SELECT
        'ARReserveInvoice',
        CAST(I.DocEntry AS NVARCHAR(50)),
        A.ActionType,
        'TEST-TIF',
        'TIF',
        'N',
        0,
        GETDATE()
    FROM inserted I
    LEFT JOIN deleted D
        ON I.DocEntry = D.DocEntry
    CROSS APPLY
    (
        SELECT CASE
            WHEN D.DocEntry IS NULL AND ISNULL(I.CANCELED, 'N') <> 'C' THEN 'ADD'
            WHEN I.CANCELED = 'Y' AND ISNULL(D.CANCELED, 'N') <> 'Y' THEN 'CANCEL'
        END AS ActionType
    ) A
    WHERE I.isIns = 'Y'
      AND A.ActionType IS NOT NULL
      AND NOT EXISTS
      (
          SELECT 1
          FROM TEMP_API.dbo.INT_OdooQueue Q
          WHERE Q.ObjectType COLLATE DATABASE_DEFAULT = 'ARReserveInvoice'
            AND Q.ObjectKey COLLATE DATABASE_DEFAULT = CAST(I.DocEntry AS NVARCHAR(50))
            AND Q.ActionType COLLATE DATABASE_DEFAULT = A.ActionType COLLATE DATABASE_DEFAULT
            AND Q.SiteId COLLATE DATABASE_DEFAULT IN ('TEST-TIF', 'TEST')
            AND Q.CompanyName COLLATE DATABASE_DEFAULT = 'TIF'
            AND Q.Status COLLATE DATABASE_DEFAULT IN ('N', 'P')
      );
END;
GO

Verification
------------

SELECT
    DB_NAME() AS DatabaseName,
    name AS TriggerName,
    is_disabled
FROM sys.triggers
WHERE name IN
(
    'TRG_OWOR_ToOdooQueue',
    'TRG_OINV_Reserve_ToOdooQueue'
);

SELECT TOP 100
    QueueId,
    ObjectType,
    ObjectKey,
    ActionType,
    SiteId,
    CompanyName,
    SapDatabaseName,
    Status,
    RetryCount,
    CreateDate
FROM TEMP_API.dbo.INT_OdooQueue
ORDER BY QueueId DESC;
